# Runtime dependencies.
#
# Pins are chosen so that `uv lock` resolves on Python 3.12 (pinned via the
# repo-root pyproject.toml's requires-python, since Vercel's own default has
# since moved past what numpy/pandas ship prebuilt cp3xx wheels for) and every
# package has a prebuilt manylinux_2_28 cp312 wheel, so nothing has to be
# built from source during the Vercel build.
#
# Dev/test-only tooling lives in requirements-dev.txt so it is not shipped in
# the deployed function bundle.

# Web framework — app.py builds a Flask app via create_app()
flask==3.1.2
flask-cors==6.0.1
flask-limiter==3.12.0
flask-caching==2.3.1
werkzeug==3.1.3

# WSGI server used by run.py
waitress==3.0.2

# Supabase (Postgres + Auth) — replaces firebase-admin
supabase==2.15.1

# Environment and Configuration
python-dotenv==1.1.1

# Logging
loguru==0.7.3

# Data Processing
# Keep numpy >=2.0 in step with pandas: pandas 2.x on Python 3.12 requires
# numpy>=1.26, and the pinned scikit-learn below was built against numpy 2.x.
pandas==2.3.2
numpy==2.3.3

# ML libraries — dosha_estimator.py / train_dosha_model.py
# scikit-learn MUST stay at 1.7.2: dosha_model.pkl was pickled with 1.7.2 and
# loading it under a different minor version raises InconsistentVersionWarning
# and risks silently wrong predictions. Retrain the model before bumping this.
scikit-learn==1.7.2
joblib==1.5.2

# XGBoost — the maternal anaemia / pregnancy-risk models in
# disease_detection/ml. These are stored in XGBoost's native JSON format
# (anemia_status_xgb.json, pregnancy_risk_xgb.json), which loads across XGBoost
# versions, so this pin can be bumped without retraining. If a model fails to
# load, the pipeline falls back to its rule-based baseline detectors.
xgboost==3.2.0

# OpenAI
openai==1.108.1

# HTTP Requests
httpx==0.28.1
requests==2.32.5

# Validation
# Pinned to a release recent enough to ship a prebuilt cp314 wheel for
# pydantic-core, so Vercel's build never has to compile it from source via
# Rust/PyO3 (older pins fail there once the build image bumps its default
# Python past what that PyO3 release supports).
pydantic==2.13.4
